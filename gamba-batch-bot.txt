    function getTargetMultiplier() {

        const input =
            $("glb-win-target");

        if (!input)
            return 0;

        const value =
            Number(input.value);

        if (
            !Number.isFinite(value) ||
            value < 0
        ) {
            return 0;
        }

        return value;
    }

    function shouldPrintWin(
        multiplier,
        game
    ) {

        const m =
            Number(multiplier);

        if (!Number.isFinite(m))
            return false;

        const target =
            getTargetMultiplier();

        if (
            game === "plinko"
        ) {

            if (m < 1)
                return false;

            if (target <= 0)
                return true;

            return m >= target;
        }

        if (m <= 0)
            return false;

        if (target <= 0)
            return true;

        return m >= target;
    }

    // =========================================================
    // FORMAT NUMBER
    // =========================================================

    function formatNumber(
        value,
        decimals = 8
    ) {

        const n =
            Number(value);

        if (!Number.isFinite(n))
            return "0";

        if (n === 0)
            return "0";

        return n.toFixed(decimals)
            .replace(/\.?0+$/, "");
    }

    // =========================================================
    // FORMAT P/L
    // =========================================================

    function formatProfit(value) {

        const n =
            Number(value);

        if (!Number.isFinite(n))
            return `0 ${currency}`;

        const abs =
            Math.abs(n);

        const formatted =
            abs
                .toFixed(8)
                .replace(/\.?0+$/, "");

        if (n > 0)
            return `+${formatted} ${currency}`;

        if (n < 0)
            return `-${formatted} ${currency}`;

        return `0 ${currency}`;
    }

    function renderWinningLogs() {

                document.createElement("div");

            empty.className =
                "glb-empty";

                empty
            );

            return;
        }

            getTargetMultiplier();

        const visible =
            winningLogs
                .filter(win => {

                    const m =
                        Number(win.multiplier);

                    if (!Number.isFinite(m))
                        return false;

                    if (
                        win.game === "plinko"
                    ) {

                        if (m < 1)
                            return false;

                        if (target <= 0)
                            return true;

                        return m >= target;
                    }

                    if (m <= 0)
                        return false;

                    if (target <= 0)
                        return true;

                    return m >= target;
                })
                .slice(
                    -MAX_VISIBLE_WINS
                );

        if (!visible.length) {

            const empty =
                document.createElement("div");

            empty.className =
                "glb-empty";

            empty.textContent =
                "No winning bets match target";

            container.appendChild(
                empty
            );

            return;
        }

        const fragment =
            document.createDocumentFragment();

        for (
            const win of visible
        ) {

            const row =
                document.createElement("div");

            row.className =
                "glb-win";

            const number =
                document.createElement("span");

            number.className =
                "glb-win-number";

            number.textContent =
                `#${win.number}`;

            const nonce =
                document.createElement("span");

            nonce.className =
                "glb-win-nonce";

                String(win.nonce);

                document.createElement("span");

                "glb-win-multiplier";

            row.appendChild(number);
            row.appendChild(nonce);

            row.appendChild(payout);

            container.scrollHeight;

    // =========================================================
    // DECOMPRESS RESULTS
    // =========================================================

    async function decodeResults(
        compressed
    ) {

        try {

            if (
                typeof compressed !==
                "string"
            ) {
                return null;
            }

            const binary =
                atob(compressed);

            const bytes =
                new Uint8Array(
                    binary.length
                );

            for (
                let i = 0;
                i < binary.length;
                i++
            ) {

                bytes[i] =
                    binary.charCodeAt(i);
            }

            const stream =
                new Blob([bytes])
                    .stream()
                    .pipeThrough(
                        new DecompressionStream(
                            "gzip"
                        )
                    );

            const buffer =
                await new Response(
                    stream
                ).arrayBuffer();

            const text =
                new TextDecoder()
                    .decode(buffer);

            return JSON.parse(text);

        } catch (err) {

            console.error(
                "[GAMBA] Batch decode error:",
                err
            );

            return null;
        }
    }

    // =========================================================
    // DETERMINE RESULT MULTIPLIER
    // =========================================================

    function getResultMultiplier(
        result
    ) {

        if (!result)
            return 0;

        const possible = [
            result.m,
            result.multiplier,
            result.payoutMultiplier,
            result.x
        ];

        for (
            const value of possible
        ) {

            const n =
                Number(value);

            if (
                Number.isFinite(n)
            ) {
                return n;
            }
        }

        return 0;
    }

    // =========================================================
    // DETERMINE NONCE
    // =========================================================

    function getResultNonce(
        result,
        index
    ) {

        if (!result)
            return index + 1;

        const possible = [
            result.n,
            result.nonce,
            result.round
        ];

        for (
            const value of possible
        ) {

            if (
                value !== undefined &&
                value !== null
            ) {

                return value;
            }
        }

        return index + 1;
    }

    // =========================================================
    // CREATE BATCH ID
    // =========================================================

    function getBatchId(
        batch,
        game
    ) {

        if (!batch)
            return null;

        const directIds = [
            batch.id,
            batch.betId,
            batch.batchId,
            batch.bet_id,
            batch.batch_id
        ];

        for (
            const id of directIds
        ) {

            if (
                id !== undefined &&
                id !== null &&
                String(id) !== ""
            ) {

                return `${game}:id:${String(id)}`;
            }
        }

        if (
            batch.resultsCompressed
        ) {

            return [
                game,
                batch.resultsCompressed,
                batch.numberOfRounds,
                batch.totalBetAmount,
                batch.totalPayout
            ].join("|");
        }

        const signature = [
            game,
            batch.numberOfRounds,
            batch.totalBetAmount,
            batch.totalPayout,
            batch.target,
            batch.betAmountPerRound,
            batch.currencyCode,
            batch.status
        ].join("|");

        return signature;
    }

    // =========================================================
    // CHECK DUPLICATE BATCH
    // =========================================================

    function alreadyProcessedBatch(
        batch,
        game
    ) {

        const id =
            getBatchId(
                batch,
                game
            );

        if (!id)
            return false;

        if (
            processedBatches.has(id)
        ) {

            console.log(
                "%c[GAMBA] Duplicate batch ignored",
                "color:#ffaa00",
                game
            );

            return true;
        }

        processedBatches.add(id);

        if (
            processedBatches.size >
            MAX_PROCESSED_BATCHES
        ) {

            const first =
                processedBatches.values()
                    .next()
                    .value;

            if (first !== undefined) {

                processedBatches.delete(
                    first
                );
            }
        }

        return false;
    }

    // =========================================================
    // RECORD SINGLE ROUND RESULT (STREAK UPDATES)
    // =========================================================

    function recordRoundOutcome(isWin) {
        if (isWin) {
            currentWinStreak++;
            currentLoseStreak = 0;

            if (currentWinStreak > longestWinStreak) {
                longestWinStreak = currentWinStreak;
            }
        } else {
            currentLoseStreak++;
            currentWinStreak = 0;

            if (currentLoseStreak > longestLoseStreak) {
                longestLoseStreak = currentLoseStreak;
            }
        }
    }

    // =========================================================
    // PROCESS LIMBO
    // =========================================================

    async function processLimbo(
        batch
    ) {

        const results =
            await decodeResults(
                batch.resultsCompressed
            );

        if (!results) {

            $("glb-status").textContent =
                "Failed to decode Limbo batch";

            return;
        }

        const target =
            Number(batch.target);

        let wins = 0;
        let losses = 0;

        let batchWagered =
            Number(batch.totalBetAmount);

        if (
            !Number.isFinite(batchWagered) ||
            batchWagered < 0
        ) {

            batchWagered = 0;
        }

        const batchPayout =
            Number(batch.totalPayout) || 0;

        let reachedLoseStreakStop = false;

        for (
            let i = 0;
            i < results.length;
            i++
        ) {

            const r =
                results[i];

            const multiplier =
                getResultMultiplier(r);

            const isWin = Number.isFinite(target) && multiplier >= target;

            recordRoundOutcome(isWin);

            if (isWin) {

                wins++;

                addWinningLog(
                    getResultNonce(
                        r,
                        i
                    ),
                    multiplier,
                    r.p,
                    "limbo"
                );

            } else {

                losses++;

                if (stopOnLoseStreak && stopOnLoseStreakN > 0 && currentLoseStreak >= stopOnLoseStreakN) {
                    reachedLoseStreakStop = true;
                }
            }
        }

        applyBatchStats(
            batch,
            results.length,
            wins,
            losses,
            batchWagered,
            batchPayout
        );

        const last =
            results[
                results.length - 1
            ];

        if (last) {

            updateLastResult(
                getResultMultiplier(last),
                target,
                results.length,
                batch.numberOfRounds
            );
        }

        $("glb-status").textContent =
            `✓ LIMBO • ${results.length} bets • ${wins} wins`;

        /*
         * STOP TRIGGERS
         */
        if (
            stopOnWin &&
            wins > 0 &&
            autoBetRunning
        ) {

            stopAutoBet(
                `WIN detected in latest Limbo batch (${wins} win${wins === 1 ? "" : "s"})`
            );
        } else if (
            stopOnLoseStreak &&
            reachedLoseStreakStop &&
            autoBetRunning
        ) {

            stopAutoBet(
                `Lose streak reached limit of ${stopOnLoseStreakN} (Current: ${currentLoseStreak})`
            );
        }
    }

    // =========================================================
    // GENERIC BATCH
    // =========================================================

    async function processGenericBatch(
        batch,
        game
    ) {

        const results =
            await decodeResults(
                batch.resultsCompressed
            );

        let resultCount =
            Array.isArray(results)
                ? results.length
                : Number(batch.numberOfRounds) || 0;

        let wins = 0;
        let losses = 0;

        let batchWagered =
            Number(batch.totalBetAmount);

        if (
            !Number.isFinite(batchWagered) ||
            batchWagered <= 0
        ) {

            const perRound =
                Number(batch.betAmountPerRound);

            if (
                Number.isFinite(perRound) &&
                perRound > 0
            ) {

                batchWagered =
                    perRound *
                    Number(batch.numberOfRounds);

            } else {

                batchWagered = 0;
            }
        }

        const batchPayout =
            Number(batch.totalPayout) || 0;

        let reachedLoseStreakStop = false;

        // -----------------------------------------------------
        // DECODED RESULTS
        // -----------------------------------------------------

        if (
            Array.isArray(results) &&
            results.length
        ) {

            for (
                let i = 0;
                i < results.length;
                i++
            ) {

                const r =
                    results[i];

                const multiplier =
                    getResultMultiplier(r);

                let isWin;

                if (
                    game === "plinko"
                ) {

                    isWin =
                        multiplier >= 1;

                } else {

                    isWin =
                        multiplier > 0;
                }

                recordRoundOutcome(isWin);

                if (isWin) {

                    wins++;

                    addWinningLog(
                        getResultNonce(
                            r,
                            i
                        ),
                        multiplier,
                        r?.p ??
                        r?.payout ??
                        0,
                        game
                    );

                } else {

                    losses++;

                    if (stopOnLoseStreak && stopOnLoseStreakN > 0 && currentLoseStreak >= stopOnLoseStreakN) {
                        reachedLoseStreakStop = true;
                    }
                }
            }

        } else {

            const isWinBatch = batch.status === "won";

            for (let i = 0; i < resultCount; i++) {

                recordRoundOutcome(isWinBatch);

                if (isWinBatch) {
                    wins++;
                } else {
                    losses++;
                    if (stopOnLoseStreak && stopOnLoseStreakN > 0 && currentLoseStreak >= stopOnLoseStreakN) {
                        reachedLoseStreakStop = true;
                    }
                }
            }
        }

        if (
            wins + losses <
            resultCount
        ) {

            const missingLosses = resultCount - wins - losses;

            for (let i = 0; i < missingLosses; i++) {

                recordRoundOutcome(false);

                losses++;

                if (stopOnLoseStreak && stopOnLoseStreakN > 0 && currentLoseStreak >= stopOnLoseStreakN) {
                    reachedLoseStreakStop = true;
                }
            }
        }

        applyBatchStats(
            batch,
            resultCount,
            wins,
            losses,
            batchWagered,
            batchPayout
        );

        let lastMultiplier = null;

        if (
            Array.isArray(results) &&
            results.length
        ) {

            lastMultiplier =
                getResultMultiplier(
                    results[
                        results.length - 1
                    ]
                );
        }

        if (
            Number.isFinite(lastMultiplier)
        ) {

            updateLastResult(
                lastMultiplier,
                Number(batch.target),
                resultCount,
                batch.numberOfRounds
            );

        } else {

            $("glb-last").textContent =
                game.toUpperCase();

            $("glb-last").style.color =
                "#fff";

            $("glb-last-label").textContent =
                `${resultCount} / ${batch.numberOfRounds} rounds`;
        }

        $("glb-status").textContent =
            `✓ ${game.toUpperCase()} • ${resultCount} bets • ${wins} wins`;

        /*
         * STOP TRIGGERS
         */
        if (
            stopOnWin &&
            wins > 0 &&
            autoBetRunning
        ) {

            stopAutoBet(
                `WIN detected in latest ${game.toUpperCase()} batch (${wins} win${wins === 1 ? "" : "s"})`
            );
        } else if (
            stopOnLoseStreak &&
            reachedLoseStreakStop &&
            autoBetRunning
        ) {

            stopAutoBet(
                `Lose streak reached limit of ${stopOnLoseStreakN} (Current: ${currentLoseStreak})`
            );
        }
    }

    // =========================================================
    // APPLY BATCH STATS
    // =========================================================

    function applyBatchStats(
        batch,
        rounds,
        wins,
        losses,
        wagered,
        payout
    ) {

        const safeRounds =
            Number.isFinite(
                Number(rounds)
            )
                ? Number(rounds)
                : 0;

        const safeWins =
            Number.isFinite(
                Number(wins)
            )
                ? Number(wins)
                : 0;

        const safeLosses =
            Number.isFinite(
                Number(losses)
            )
                ? Number(losses)
                : 0;

        const safeWagered =
            Number.isFinite(
                Number(wagered)
            )
                ? Number(wagered)
                : 0;

        const safePayout =
            Number.isFinite(
                Number(payout)
            )
                ? Number(payout)
                : 0;

        const batchProfit =
            safePayout -
            safeWagered;

        totalBets +=
            safeRounds;

        totalWins +=
            safeWins;

        totalLosses +=
            safeLosses;

        totalWagered +=
            safeWagered;

        totalProfit +=
            batchProfit;

        if (
            batch.currencyCode
        ) {

            currency =
                String(
                    batch.currencyCode
                );
        }

        $("glb-target").textContent =
            batch.target !== undefined &&
            batch.target !== null
                ? `${batch.target}x`
                : "—";

        renderStats();

        saveStats();

        console.log(
            "%c[GAMBA STATS]",
            "color:#55ff88;font-weight:bold",
            {
                batchWagered:
                    safeWagered,

                batchPayout:
                    safePayout,

                batchProfit,

                batchWins:
                    safeWins,

                batchLosses:
                    safeLosses,

                currentWinStreak,
                longestWinStreak,
                currentLoseStreak,
                longestLoseStreak,

                totalBets,
                totalWagered,
                totalProfit
            }
        );
    }

    // =========================================================
    // PROCESS BATCH
    // =========================================================

    async function processBatch(
        batch,
        game
    ) {

        if (!batch)
            return;

        if (
            alreadyProcessedBatch(
                batch,
                game
            )
        ) {

            return;
        }

        console.log(
            "%c[GAMBA BATCH]",
            "color:#55ff88;font-weight:bold",
            game,
            batch
        );

        $("glb-status").textContent =
            `Processing ${game} batch...`;

        if (
            batch.currencyCode
        ) {

            currency =
                String(
                    batch.currencyCode
                );
        }

        if (
            game === "limbo"
        ) {

            await processLimbo(
                batch
            );

        } else {

            await processGenericBatch(
                batch,
                game
            );
        }
    }

    // =========================================================
    // INSPECT
    // =========================================================

    const BATCH_KEYS = [

        "limboBatchBet",
        "minesBatchBet",
        "diceBatchBet",
        "plinkoBatchBet",
        "kenoBatchBet",
        "dragonTowerBatchBet",
        "pumpBatchBet"
    ];

    function inspect(
        obj
    ) {

        if (
            !obj ||
            typeof obj !== "object"
        ) {

            return;
        }

        for (
            const key of BATCH_KEYS
        ) {

            const batch =
                obj?.data?.[key] ||
                obj?.[key];

            if (batch) {

                const game =
                    key
                        .replace(
                            "BatchBet",
                            ""
                        );

                processBatch(
                    batch,
                    game
                );
            }
        }

        for (
            const key of Object.keys(obj)
        ) {

            try {

                const value =
                    obj[key];

                if (
                    value &&
                    typeof value ===
                    "object"
                ) {

                    inspect(value);
                }

            } catch {}
        }
    }

    // =========================================================
    // INSPECT TEXT
    // =========================================================

    function inspectText(
        text
    ) {

        if (
            typeof text !==
            "string"
        ) {

            return;
        }

        if (
            !BATCH_KEYS.some(
                key =>
                    text.includes(key)
            )
        ) {

            return;
        }

        try {

            inspect(
                JSON.parse(text)
            );

        } catch {}
    }

    // =========================================================
    // FETCH
    // =========================================================

    const originalFetch =
        window.fetch;

    window.fetch =
        async function(...args) {

            const response =
                await originalFetch.apply(
                    this,
                    args
                );

            try {

                const clone =
                    response.clone();

                clone.text()
                    .then(
                        inspectText
                    )
                    .catch(
                        () => {}
                    );

            } catch {}

            return response;
        };

    // XHR
    // =========================================================

    const oldOpen =
        XMLHttpRequest.prototype.open;

    const oldSend =
        XMLHttpRequest.prototype.send;

    XMLHttpRequest.prototype.open =
        function(
            method,
            url,
            ...rest
        ) {

            this.__gambaURL =
                url;

            return oldOpen.call(
                this,
                method,
                url,
                ...rest
            );
        };

    XMLHttpRequest.prototype.send =
        function(...args) {

            this.addEventListener(
                "load",
                () => {

                    try {

                        inspectText(
                            this.responseText
                        );

                    } catch {}

                }
            );

            return oldSend.apply(
                this,
                args
            );
        };

    // =========================================================
    // WEBSOCKET
    // =========================================================

    const OriginalWebSocket =
        window.WebSocket;

    window.WebSocket =
        new Proxy(
            OriginalWebSocket,
            {

                construct(
                    Target,
                    args
                ) {

                    const ws =
                        new Target(...args);

                    activeSocket =
                        ws;

                    ws.addEventListener(
                        "message",
                        event => {

                            if (
                                typeof event.data ===
                                "string"
                            ) {

                                inspectText(
                                    event.data
                                );

                            } else if (
                                event.data instanceof Blob
                            ) {

                                event.data
                                    .text()
                                    .then(
                                        inspectText
                                    )
                                    .catch(
                                        () => {}
                                    );
                            }
                        }
                    );

                    ws.addEventListener(
                        "close",
                        () => {

                            if (
                                activeSocket ===
                                ws
                            ) {

                                activeSocket =
                                    null;

                                $("glb-status").textContent =
                                    "WebSocket disconnected";
                            }
                        }
                    );

                    return ws;
                }
            }
        );

    // =========================================================
    // MINIMIZE
    // =========================================================

    $("glb-min").addEventListener(
        "click",
        e => {

            e.stopPropagation();

            minimized =
                !minimized;

            ui.classList.toggle(
                "glb-minimized",
                minimized
            );

            $("glb-min").textContent =
                minimized
                    ? "+"
                    : "−";
        }
    );

    // =========================================================
    // TOUCH / MOUSE DRAG
    // =========================================================

    let dragging = false;

    let dragOffsetX = 0;
    let dragOffsetY = 0;

    function startDrag(
        clientX,
        clientY
    ) {

        const rect =
            ui.getBoundingClientRect();

        dragOffsetX =
            clientX -
            rect.left;

        dragOffsetY =
            clientY -
            rect.top;

        ui.style.left =
            `${rect.left}px`;

        ui.style.top =
            `${rect.top}px`;

        ui.style.right =
            "auto";

        dragging = true;
    }

    function moveDrag(
        clientX,
        clientY
    ) {

        if (!dragging)
            return;

        const width =
            ui.offsetWidth;

        const height =
            ui.offsetHeight;

        let left =
            clientX -
            dragOffsetX;

        let top =
            clientY -
            dragOffsetY;

        const maxLeft =
            window.innerWidth -
            width;

        const maxTop =
            window.innerHeight -
            height;

        left =
            Math.max(
                0,
                Math.min(
                    left,
                    maxLeft
                )
            );

        top =
            Math.max(
                0,
                Math.min(
                    top,
                    maxTop
                )
            );

        ui.style.left =
            `${left}px`;

        ui.style.top =
            `${top}px`;
    }

    function stopDrag() {

        dragging = false;
    }

    const header =
        $("glb-head");

    // =========================================================
    // MOUSE DRAG
    // =========================================================

    header.addEventListener(
        "mousedown",
        e => {

            if (
                e.target ===
                $("glb-min")
            ) {

                return;
            }

            startDrag(
                e.clientX,
                e.clientY
            );

            e.preventDefault();
        }
    );

    document.addEventListener(
        "mousemove",
        e => {

            moveDrag(
                e.clientX,
                e.clientY
            );
        }
    );

    document.addEventListener(
        "mouseup",
        stopDrag
    );

    // =========================================================
    // TOUCH DRAG
    // =========================================================

    header.addEventListener(
        "touchstart",
        e => {

            if (
                e.target ===
                $("glb-min")
            ) {

                return;
            }

            const touch =
                e.touches[0];

            if (!touch)
                return;

            startDrag(
                touch.clientX,
                touch.clientY
            );

            e.preventDefault();

        },
        {
            passive: false
        }
    );

    document.addEventListener(
        "touchmove",
        e => {

            if (!dragging)
                return;

            const touch =
                e.touches[0];

            if (!touch)
                return;

            moveDrag(
                touch.clientX,
                touch.clientY
            );

            e.preventDefault();

        },
        {
            passive: false
        }
    );

    document.addEventListener(
        "touchend",
        stopDrag
    );

    document.addEventListener(
        "touchcancel",
        stopDrag
    );

    // =========================================================
    // INITIALIZE

    renderStats();
    renderWinningLogs();

    updateStopOnWinUI();
    updateStopOnLoseStreakUI();

    console.log(
        "%c[GAMBA ADVANCED BATCH MONITOR READY]",
        "color:#55ff88;font-size:16px;font-weight:bold"
    );

})();
